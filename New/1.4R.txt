#include <Wire.h>
#include <Adafruit_GFX.h>
#include <Adafruit_SSD1306.h>

#define SCREEN_WIDTH 128
#define SCREEN_HEIGHT 64
Adafruit_SSD1306 display(SCREEN_WIDTH, SCREEN_HEIGHT, &Wire, -1);

// --- Pines ---
const int PIN_OK = 2;
const int PIN_ESC = 3;
const int PIN_L = 7;
const int PIN_R = 8;
const int JOY_X = A0;
const int JOY_Y = A1;

// --- Configuración Global ---
float sensibilidad = 4.0;
int brillo = 150;
float cursorX = 64, cursorY = 32;
int currentApp = -1;
int selectedIcon = -1;

// --- Sistema de Pestañas e Indicador Simétrico ---
int currentTab = 0;
float scrollX = 0;
float targetScrollX = 0;
const float suavizado = 0.25;
int showTabIndicator = 0;          
unsigned long tabIndicatorTime = 0; 
const unsigned long indicatorDuration = 720; // Duración total optimizada para Entrada, Centro y Salida

// --- Estructura Global para Evitar Bloqueos del Compilador ---
struct Btn {
  int x;
  int y;
  int w;
  int h;
  char val;
  const char* label;
};

// --- Prototipos de Funciones ---
void bootSystem();
void updateDesktop();
void drawDesktop();
void runFullCalc();
void runAdvancedTimer();
void runSettings();
void runProximamente();
void runStickGraph();

void setup() {
  delay(300); 
  pinMode(PIN_OK, INPUT_PULLUP);
  pinMode(PIN_ESC, INPUT_PULLUP);
  pinMode(PIN_L, INPUT_PULLUP);
  pinMode(PIN_R, INPUT_PULLUP);

  if(!display.begin(SSD1306_SWITCHCAPVCC, 0x3C)) {
    for(;;); 
  }
  
  display.setTextWrap(false);
  display.ssd1306_command(0x81); 
  display.ssd1306_command(brillo); 
  
  bootSystem();
}

void loop() {
  if (currentApp == -1) {
    updateDesktop();
    drawDesktop();
  } else {
    switch(currentApp) {
      case 0: runFullCalc(); break;
      case 1: runAdvancedTimer(); break;
      case 2: runSettings(); break;
      case 3:
      case 4: runProximamente(); break;
    }
  }
}

void bootSystem() {
  for(int frame = 0; frame < 10; frame++) {
    display.clearDisplay();
    display.setTextSize(2);
    display.setTextColor(SSD1306_WHITE);
    display.setCursor(5, 15);
    display.print(F("Arduino OS"));
    for(int i=0; i<8; i++) {
      float angle = (frame * 0.5) + (i * 0.8);
      display.fillCircle(64 + cos(angle)*10, 48 + sin(angle)*10, 1, SSD1306_WHITE);
    }
    display.display();
    delay(100);
  }
}

void updateDesktop() {
  if (digitalRead(PIN_R) == LOW && currentTab < 1) { 
    currentTab = 1; 
    targetScrollX = -128; 
    showTabIndicator = 2; 
    tabIndicatorTime = millis(); 
    delay(250); 
  }
  if (digitalRead(PIN_L) == LOW && currentTab > 0) { 
    currentTab = 0; 
    targetScrollX = 0; 
    showTabIndicator = 1; 
    tabIndicatorTime = millis(); 
    delay(250); 
  }

  scrollX += (targetScrollX - scrollX) * suavizado;

  int vx = analogRead(JOY_X);
  int vy = analogRead(JOY_Y);
  
  if(abs(vx - 512) > 50) {
    if(vx < 400) cursorX += sensibilidad;       
    else if(vx > 600) cursorX -= sensibilidad;  
  }
  if(abs(vy - 512) > 50) {
    if(vy < 400) cursorY += sensibilidad;
    else if(vy > 600) cursorY -= sensibilidad;
  }
  
  cursorX = constrain(cursorX, 0, 120);
  cursorY = constrain(cursorY, 0, 56);

  selectedIcon = -1;
  if (cursorY > 5 && cursorY < 40) {
    if (currentTab == 0) {
      if (cursorX < 42) selectedIcon = 0;
      else if (cursorX < 84) selectedIcon = 1;
      else selectedIcon = 2;
    } else {
      if (cursorX < 64) selectedIcon = 3;
      else selectedIcon = 4;
    }
  }

  if (digitalRead(PIN_OK) == LOW && selectedIcon != -1) { currentApp = selectedIcon; delay(300); }
}

void drawDesktop() {
  display.clearDisplay();
  display.setTextSize(1);
  display.setTextColor(SSD1306_WHITE);
  
  int off0 = (int)scrollX;
  int off1 = (int)scrollX + 128;

  // ================= PESTAÑA 0 =================
  if(selectedIcon == 0) display.drawRect(8+off0, 8, 28, 28, SSD1306_WHITE);
  int cX = 12 + off0; int cY = 12;
  display.drawRect(cX, cY, 20, 20, SSD1306_WHITE);
  display.fillRect(cX+3, cY+3, 14, 4, SSD1306_WHITE); 
  display.drawPixel(cX+5, cY+11, SSD1306_WHITE); display.drawPixel(cX+10, cY+11, SSD1306_WHITE); display.drawPixel(cX+15, cY+11, SSD1306_WHITE);
  display.drawPixel(cX+5, cY+15, SSD1306_WHITE); display.drawPixel(cX+10, cY+15, SSD1306_WHITE); display.drawPixel(cX+15, cY+15, SSD1306_WHITE);
  display.setCursor(14+off0, 39); display.print(F("Calc"));

  if(selectedIcon == 1) display.drawRect(50+off0, 8, 28, 28, SSD1306_WHITE);
  int tX = 54 + off0; int tY = 12;
  display.drawCircle(tX+10, tY+10, 8, SSD1306_WHITE);
  display.fillRect(tX+9, tY, 3, 2, SSD1306_WHITE); 
  display.drawLine(tX+10, tY+10, tX+10, tY+5, SSD1306_WHITE); 
  display.drawLine(tX+10, tY+10, tX+14, tY+10, SSD1306_WHITE); 
  display.setCursor(53+off0, 39); display.print(F("Time"));

  if(selectedIcon == 2) display.drawRect(92+off0, 8, 28, 28, SSD1306_WHITE);
  int gX = 96 + off0; int gY = 12;
  display.drawCircle(gX+10, gY+10, 6, SSD1306_WHITE);
  display.drawCircle(gX+10, gY+10, 2, SSD1306_WHITE);
  display.drawLine(gX+10, gY+2, gX+10, gY+4, SSD1306_WHITE);   display.drawLine(gX+10, gY+16, gX+10, gY+18, SSD1306_WHITE);
  display.drawLine(gX+2, gY+10, gX+4, gY+10, SSD1306_WHITE);   display.drawLine(gX+16, gY+10, gX+18, gY+10, SSD1306_WHITE);
  display.drawPixel(gX+5, gY+5, SSD1306_WHITE);   display.drawPixel(gX+15, gY+15, SSD1306_WHITE);
  display.drawPixel(gX+15, gY+5, SSD1306_WHITE);  display.drawPixel(gX+5, gY+15, SSD1306_WHITE);
  display.setCursor(95+off0, 39); display.print(F("Conf"));

  // ================= PESTAÑA 1 =================
  if(selectedIcon == 3) display.drawRect(20+off1, 8, 28, 28, SSD1306_WHITE);
  int jX = 24 + off1; int jY = 12;
  display.drawRect(jX+2, jY, 16, 20, SSD1306_WHITE);
  display.drawRect(jX+5, jY+3, 10, 7, SSD1306_WHITE); 
  display.drawLine(jX+4, jY+14, jX+8, jY+14, SSD1306_WHITE); 
  display.drawLine(jX+6, jY+12, jX+6, jY+16, SSD1306_WHITE); 
  display.drawPixel(jX+14, jY+13, SSD1306_WHITE); display.drawPixel(jX+12, jY+15, SSD1306_WHITE);
  display.setCursor(20+off1, 39); display.print(F("Juegos"));

  if(selectedIcon == 4) display.drawRect(80+off1, 8, 28, 28, SSD1306_WHITE);
  int cR = 84 + off1; int cRy = 12;
  display.drawRect(cR+2, cRy, 16, 19, SSD1306_WHITE);
  display.drawRect(cR+5, cRy+3, 10, 8, SSD1306_WHITE); 
  display.drawLine(cR+4, cRy+16, cR+16, cRy+16, SSD1306_WHITE);
  display.drawLine(cR+6, cRy+14, cR+14, cRy+14, SSD1306_WHITE);
  display.setCursor(78+off1, 39); display.print(F("Cartuc"));

  // ================= INDICADOR COMPLETO: FADE IN -> NORMAL -> FADE OUT =================
  if (showTabIndicator > 0) {
    unsigned long elapsed = millis() - tabIndicatorTime;
    
    if (elapsed < indicatorDuration) {
      int indX = 54, indY = 48, indW = 20, indH = 14;
      int density = 0; 
      int textColor = SSD1306_BLACK;

      // Control preciso de los intermediarios en base al tiempo transcurrido
      if (elapsed < 60)       { density = 1; textColor = SSD1306_WHITE; } // Entrada ultra sutil (12%)
      else if (elapsed < 120) { density = 2; textColor = SSD1306_WHITE; } // Entrada media (25%)
      else if (elapsed < 180) { density = 3; textColor = SSD1306_BLACK; } // Entrada ajedrez (50%)
      else if (elapsed < 240) { density = 4; textColor = SSD1306_BLACK; } // Entrada casi llena (75%)
      else if (elapsed < 480) { density = 5; textColor = SSD1306_BLACK; } // ¡APARECE NORMAL! (100% Sólido estable)
      else if (elapsed < 540) { density = 4; textColor = SSD1306_BLACK; } // Salida casi llena (75%)
      else if (elapsed < 600) { density = 3; textColor = SSD1306_BLACK; } // Salida ajedrez (50%)
      else if (elapsed < 660) { density = 2; textColor = SSD1306_WHITE; } // Salida media (25%)
      else                    { density = 1; textColor = SSD1306_WHITE; } // Salida final rala (12%)

      // Renderizado físico del nivel de degradado
      if (density == 5) {
        display.fillRoundRect(indX, indY, indW, indH, 2, SSD1306_WHITE);
      } else {
        for (int x = indX; x < indX + indW; x++) {
          for (int y = indY; y < indY + indH; y++) {
            if (density == 4 && !(x % 2 == 0 && y % 2 == 0)) display.drawPixel(x, y, SSD1306_WHITE);
            if (density == 3 && (x + y) % 2 == 0) display.drawPixel(x, y, SSD1306_WHITE);
            if (density == 2 && x % 2 == 0 && y % 2 == 0) display.drawPixel(x, y, SSD1306_WHITE);
            if (density == 1 && (x + y) % 4 == 0) display.drawPixel(x, y, SSD1306_WHITE);
          }
        }
      }

      // Imprimir el número de la pestaña con inversión de color inteligente para que sea legible
      display.setTextColor(textColor);
      display.setCursor(61, indY + 3);
      display.print(showTabIndicator);
      display.setTextColor(SSD1306_WHITE); // Reset general
    } else {
      showTabIndicator = 0; 
    }
  }
  
  display.fillTriangle(cursorX, cursorY, cursorX, cursorY+7, cursorX+5, cursorY+5, SSD1306_WHITE);
  display.display();
}

void runProximamente() {
  while(digitalRead(PIN_ESC) == HIGH) {
    display.clearDisplay();
    display.setCursor(20, 25);
    display.setTextSize(1);
    display.print(F("PROXIMAMENTE..."));
    display.display();
  }
  currentApp = -1; delay(300);
}

void runFullCalc() {
  long result = 0, currentNum = 0; char lastOp = ' '; String displayBuffer = "0";
  bool nextNumClears = false;

  static const Btn b[] = {
    {2,14,22,11,'7',"7"}, {26,14,22,11,'8',"8"}, {50,14,22,11,'9',"9"}, {76,14,22,11,'+',"+"}, {100,14,24,11,'x',"x"},
    {2,27,22,11,'4',"4"}, {26,27,22,11,'5',"5"}, {50,27,22,11,'6',"6"}, {76,27,22,11,'-',"-"}, {100,27,24,11,'/',"/"},
    {2,40,22,11,'1',"1"}, {26,40,22,11,'2',"2"}, {50,40,22,11,'3',"3"}, {76,40,22,11,'=',"="}, {100,40,24,11,'C',"C"},
    {2,53,70,10,'0',"0"}
  };

  while(digitalRead(PIN_ESC) == HIGH) {
    display.clearDisplay(); display.drawRect(0, 0, 128, 13, SSD1306_WHITE);
    display.setCursor(5, 3); display.print(displayBuffer);
    
    for(int i=0; i<16; i++) {
      display.drawRect(b[i].x, b[i].y, b[i].w, b[i].h, SSD1306_WHITE);
      int paddingX = (b[i].w - 6) / 2;
      if(b[i].val == '0') paddingX = 32; 
      
      display.setCursor(b[i].x + paddingX, b[i].y + 2); 
      display.print(b[i].label);
    }
    
    int vx = analogRead(JOY_X); int vy = analogRead(JOY_Y);
    if(abs(vx-512)>50) cursorX += (vx > 600 ? -sensibilidad : sensibilidad);
    if(abs(vy-512)>50) cursorY += (vy < 400 ? sensibilidad : -sensibilidad);
    cursorX = constrain(cursorX, 0, 120); cursorY = constrain(cursorY, 0, 60);
    
    display.fillCircle(cursorX, cursorY, 1, SSD1306_WHITE);
    
    if(digitalRead(PIN_OK) == LOW) {
      for(int i=0; i<16; i++) {
        if(cursorX >= b[i].x && cursorX <= b[i].x+b[i].w && cursorY >= b[i].y && cursorY <= b[i].y+b[i].h) {
          char val = b[i].val;
          if(val >= '0' && val <= '9') {
            if(nextNumClears) { currentNum = 0; nextNumClears = false; }
            currentNum = (currentNum * 10) + (val - '0');
            displayBuffer = String(currentNum);
          } else if(val == '+' || val == '-' || val == 'x' || val == '/') {
            result = currentNum; lastOp = val; nextNumClears = true;
          } else if(val == '=') {
            if(lastOp == '+') result += currentNum; else if(lastOp == '-') result -= currentNum;
            else if(lastOp == 'x') result *= currentNum; else if(lastOp == '/' && currentNum != 0) result /= currentNum;
            displayBuffer = String(result); currentNum = result; nextNumClears = true;
          } else if(val == 'C') { currentNum = 0; result = 0; lastOp = ' '; displayBuffer = "0"; nextNumClears = false; }
          delay(250);
        }
      }
    }
    display.display();
  }
  currentApp = -1; delay(300);
}

void runAdvancedTimer() {
  unsigned long startT = 0, elapsed = 0; bool running = false;
  while(digitalRead(PIN_ESC) == HIGH) {
    if(running) elapsed = (millis() - startT) / 1000;
    
    display.clearDisplay(); 
    display.drawRect(0, 0, 128, 64, SSD1306_WHITE);
    
    display.drawCircle(35, 32, 22, SSD1306_WHITE); 
    display.drawCircle(35, 32, 19, SSD1306_WHITE); 
    display.fillRect(33, 6, 5, 4, SSD1306_WHITE);  
    display.drawPixel(35, 17, SSD1306_WHITE);      
    display.drawPixel(35, 47, SSD1306_WHITE);      
    
    float angle = (elapsed % 60) * 0.1047;
    display.drawLine(35, 32, 35 + cos(angle-1.57)*14, 32 + sin(angle-1.57)*14, SSD1306_WHITE);
    
    display.setTextSize(2); 
    display.setCursor(72, 16); 
    if(elapsed < 10) display.print(F("0")); 
    display.print(elapsed); 
    display.setTextSize(1); display.print(F("s"));
    
    display.drawRoundRect(70, 40, 50, 16, 3, SSD1306_WHITE);
    display.setCursor(79, 44); 
    display.print(running ? F("STOP") : F("START"));
    
    if(digitalRead(PIN_OK) == LOW) {
      if(!running) { startT = millis() - (elapsed * 1000); running = true; }
      else running = false; delay(300);
    }
    display.display();
  }
  currentApp = -1; delay(300);
}

void runSettings() {
  int subMenu = 0;
  while(digitalRead(PIN_ESC) == HIGH) {
    display.clearDisplay();
    display.setCursor(0,0); display.println(F("CONFIGURACION"));
    display.drawLine(0, 9, 128, 9, SSD1306_WHITE);
    
    const char* opts[] = {"ESPECIFICACIONES", "PRUEBA STICK", "SENSIBILIDAD", "BRILLO"};
    for(int i=0; i<4; i++) {
      display.setCursor(10, 15+(i*11));
      if(subMenu == i) display.print(F("> "));
      display.println(opts[i]);
    }
    
    int vy = analogRead(JOY_Y);
    if(vy < 300) { subMenu++; if(subMenu > 3) subMenu = 0; delay(200); }
    if(vy > 700) { subMenu--; if(subMenu < 0) subMenu = 3; delay(200); }
    
    if(digitalRead(PIN_OK) == LOW) {
      delay(200);
      if(subMenu == 0) {
        while(digitalRead(PIN_ESC) == HIGH) {
          display.clearDisplay(); display.setCursor(0,0); display.println(F("SYSTEM INFO:"));
          display.drawLine(0, 9, 128, 9, SSD1306_WHITE);
          display.println(F("OS: Arduino OS v11"));
          display.println(F("Board: Uno/Nano"));
          display.println(F("Chip: ATmega328P"));
          display.println(F("SRAM: 2048 Bytes"));
          display.display();
        }
      } else if(subMenu == 1) runStickGraph();
      else if(subMenu == 2) {
        while(digitalRead(PIN_ESC) == HIGH) {
          display.clearDisplay(); display.setCursor(0,10); display.print(F("SENS: ")); display.print(sensibilidad);
          
          int valX = analogRead(JOY_X);
          if(valX < 300) sensibilidad += 0.5; 
          if(valX > 700) sensibilidad -= 0.5;
          
          sensibilidad = constrain(sensibilidad, 1, 10);
          display.fillRect(10, 30, sensibilidad*10, 10, SSD1306_WHITE); display.display(); delay(100);
        }
      } else if(subMenu == 3) {
        while(digitalRead(PIN_ESC) == HIGH) {
          display.clearDisplay(); display.setCursor(0,10); display.print(F("BRILLO: ")); display.print(brillo);
          
          int valX = analogRead(JOY_X);
          if(valX < 300) brillo += 15; 
          if(valX > 700) brillo -= 15;
          
          brillo = constrain(brillo, 0, 255);
          
          display.ssd1306_command(0x81);
          display.ssd1306_command(brillo);
          
          display.fillRect(10, 30, map(brillo, 0, 255, 0, 100), 10, SSD1306_WHITE); display.display(); delay(100);
        }
      }
      delay(200);
    }
    display.display();
  }
  currentApp = -1; delay(300);
}

void runStickGraph() {
  while(digitalRead(PIN_ESC) == HIGH) {
    int rx = analogRead(JOY_X); int ry = analogRead(JOY_Y);
    
    int gX = map(rx, 0, 1023, 118, 10); 
    int gY = map(ry, 0, 1023, 54, 10);
    
    display.clearDisplay();
    display.drawRect(0, 0, 128, 64, SSD1306_WHITE);
    display.setCursor(4, 4); display.print(F("STICK ANALYZER"));
    display.drawLine(64, 10, 64, 54, SSD1306_WHITE); display.drawLine(10, 32, 118, 32, SSD1306_WHITE);
    
    display.drawCircle(gX, gY, 5, SSD1306_WHITE);
    
    display.setCursor(4, 55); display.print(F("X:")); display.print(rx);
    display.setCursor(70, 55); display.print(F("Y:")); display.print(ry);
    display.display();
  }
  delay(300);
}
