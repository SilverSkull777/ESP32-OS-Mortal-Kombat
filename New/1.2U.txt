#include <Wire.h>
#include <Adafruit_GFX.h>
#include <Adafruit_SSD1306.h>

#define SCREEN_WIDTH 128
#define SCREEN_HEIGHT 64
Adafruit_SSD1306 display(SCREEN_WIDTH, SCREEN_HEIGHT, &Wire, -1);

// --- Pines ---
const int PIN_OK = 2;  
const int PIN_ESC = 3;  
const int PIN_L = 7;    
const int PIN_R = 8;    
const int JOY_X = A0;
const int JOY_Y = A1;

// --- Configuración ---
float sensibilidad = 4.0; 
int brillo = 150; 
float cursorX = 64, cursorY = 32;
int currentApp = -1; 
int selectedIcon = -1;

// --- Sistema de Pestañas ---
int currentTab = 0;
float scrollX = 0;        
float targetScrollX = 0;  
const float suavizado = 0.25; 

void setup() {
  pinMode(PIN_OK, INPUT_PULLUP);
  pinMode(PIN_ESC, INPUT_PULLUP);
  pinMode(PIN_L, INPUT_PULLUP);
  pinMode(PIN_R, INPUT_PULLUP);

  if(!display.begin(SSD1306_SWITCHCAPVCC, 0x3C)) for(;;);
  display.ssd1306_command(SSD1306_SETCONTRAST);
  display.ssd1306_command(brillo); 
  bootSystem(); 
}

void loop() {
  if (currentApp == -1) {
    updateDesktop();
    drawDesktop();
  } else {
    if (currentApp == 0) runFullCalc(); 
    if (currentApp == 1) runAdvancedTimer(); 
    if (currentApp == 2) runSettings(); 
    if (currentApp == 3 || currentApp == 4) runProximamente(); 
  }
}

void bootSystem() {
  for(int frame = 0; frame < 10; frame++) {
    display.clearDisplay();
    display.setTextSize(2); display.setTextColor(1);
    display.setCursor(5, 15); display.print(F("Arduino OS"));
    for(int i=0; i<8; i++) {
      float angle = (frame * 0.5) + (i * 0.8);
      display.fillCircle(64 + cos(angle)*10, 48 + sin(angle)*10, 1, 1);
    }
    display.display(); delay(100);
  }
}

void updateDesktop() {
  // --- Cambio de Pestañas ---
  if (digitalRead(PIN_R) == LOW && currentTab < 1) { currentTab = 1; targetScrollX = -128; delay(250); }
  if (digitalRead(PIN_L) == LOW && currentTab > 0) { currentTab = 0; targetScrollX = 0; delay(250); }

  scrollX += (targetScrollX - scrollX) * suavizado;

  // Movimiento Cursor
  int vx = analogRead(JOY_X); int vy = analogRead(JOY_Y);
  if(abs(vx - 512) > 50) {
    if(vx < 400) cursorX -= sensibilidad; 
    else if(vx > 600) cursorX += sensibilidad;
  }
  if(abs(vy - 512) > 50) {
    if(vy < 400) cursorY += sensibilidad; 
    else if(vy > 600) cursorY -= sensibilidad; 
  }
  cursorX = constrain(cursorX, 0, 120); cursorY = constrain(cursorY, 0, 54);

  // Selección de Iconos (detecta cuál está bajo el cursor según el scroll)
  selectedIcon = -1;
  if (cursorY > 5 && cursorY < 40) {
    if (scrollX > -64) { // PESTAÑA 0
      if (cursorX < 42) selectedIcon = 0; 
      else if (cursorX < 84) selectedIcon = 1; 
      else selectedIcon = 2;
    } else { // PESTAÑA 1
      if (cursorX < 64) selectedIcon = 3; 
      else selectedIcon = 4;
    }
  }

  if (digitalRead(PIN_OK) == LOW && selectedIcon != -1) { currentApp = selectedIcon; delay(300); }
}

void drawDesktop() {
  display.clearDisplay();
  display.setTextSize(1);
  int off0 = (int)scrollX;
  int off1 = (int)scrollX + 128;

  // --- PESTAÑA 0: ORIGINAL ---
  if(selectedIcon == 0) display.drawRect(8+off0, 8, 28, 28, 1);
  display.drawRect(12+off0, 12, 20, 20, 1);
  display.drawLine(12+off0, 18, 32+off0, 18, 1);
  display.setCursor(14+off0, 38); display.print(F("Calc"));

  if(selectedIcon == 1) display.drawRect(50+off0, 8, 28, 28, 1);
  display.drawCircle(64+off0, 22, 10, 1);
  display.drawLine(64+off0, 22, 64+off0, 15, 1);
  display.setCursor(53+off0, 38); display.print(F("Time"));

  if(selectedIcon == 2) display.drawRect(92+off0, 8, 28, 28, 1);
  display.drawCircle(106+off0, 22, 8, 1);
  display.setCursor(95+off0, 38); display.print(F("Conf")); 

  // --- PESTAÑA 1: JUEGOS Y CARTUCHO ---
  if(selectedIcon == 3) display.drawRect(20+off1, 8, 28, 28, 1);
  display.drawRect(24+off1, 14, 20, 16, 1); // Gameboy style
  display.drawCircle(28+off1, 25, 2, 1); 
  display.setCursor(20+off1, 38); display.print(F("Juegos"));

  if(selectedIcon == 4) display.drawRect(80+off1, 8, 28, 28, 1);
  display.drawRect(84+off1, 12, 20, 20, 1);
  display.drawLine(84+off1, 15, 104+off1, 15, 1); // Cartucho lines
  display.setCursor(78+off1, 38); display.print(F("Cartuc"));

  // Barra de tareas
  display.fillRect(0, 56, 128, 8, 1);
  display.setTextColor(0); display.setCursor(2, 57); display.print(F("START"));
  display.setCursor(85, 57); display.print(F("v11.0"));
  display.setTextColor(1);
  
  // Puntero flecha
  display.fillTriangle(cursorX, cursorY, cursorX, cursorY+7, cursorX+5, cursorY+5, 1);
  display.display();
}

void runProximamente() {
  while(digitalRead(PIN_ESC) == HIGH) {
    display.clearDisplay();
    display.setCursor(20, 25);
    display.setTextSize(1);
    display.print(F("PROXIMAMENTE..."));
    display.display();
  }
  currentApp = -1; delay(300);
}

// ==========================================
// CALCULADORA (IGUAL A LA TUYA PERO LIMPIA)
// ==========================================
void runFullCalc() {
  long result = 0, currentNum = 0; char lastOp = ' '; String displayBuffer = "0";
  bool nextNumClears = false; 

  struct Btn { int x, y, w, h; char val; const char* label; };
  Btn b[] = {
    {2,14,22,11,'7',"7"}, {26,14,22,11,'8',"8"}, {50,14,22,11,'9',"9"}, {76,14,22,11,'+',"+"}, {100,14,24,11,'x',"x"},
    {2,27,22,11,'4',"4"}, {26,27,22,11,'5',"5"}, {50,27,22,11,'6',"6"}, {76,27,22,11,'-',"-"}, {100,27,24,11,'/',"/"},
    {2,40,22,11,'1',"1"}, {26,40,22,11,'2',"2"}, {50,40,22,11,'3',"3"}, {76,40,22,11,'=',"="}, {100,40,24,11,'C',"C"},
    {2,53,70,10,'0',"    0"}
  };

  while(digitalRead(PIN_ESC) == HIGH) {
    display.clearDisplay(); display.drawRect(0, 0, 128, 13, 1);
    display.setCursor(5, 3); display.print(displayBuffer);
    for(int i=0; i<16; i++) {
      display.drawRect(b[i].x, b[i].y, b[i].w, b[i].h, 1);
      display.setCursor(b[i].x+7, b[i].y+2); display.print(b[i].label);
    }
    int vx = analogRead(JOY_X); int vy = analogRead(JOY_Y);
    if(abs(vx-512)>50) cursorX += (vx > 600 ? sensibilidad : -sensibilidad);
    if(abs(vy-512)>50) cursorY += (vy < 400 ? sensibilidad : -sensibilidad);
    cursorX = constrain(cursorX, 0, 120); cursorY = constrain(cursorY, 0, 60);
    display.fillCircle(cursorX, cursorY, 1, 1);
    if(digitalRead(PIN_OK) == LOW) {
      for(int i=0; i<16; i++) {
        if(cursorX >= b[i].x && cursorX <= b[i].x+b[i].w && cursorY >= b[i].y && cursorY <= b[i].y+b[i].h) {
          char val = b[i].val;
          if(val >= '0' && val <= '9') {
            if(nextNumClears) { currentNum = 0; nextNumClears = false; }
            currentNum = (currentNum * 10) + (val - '0');
            displayBuffer = String(currentNum);
          } else if(val == '+' || val == '-' || val == 'x' || val == '/') {
            result = currentNum; lastOp = val; nextNumClears = true;
          } else if(val == '=') {
            if(lastOp == '+') result += currentNum; else if(lastOp == '-') result -= currentNum;
            else if(lastOp == 'x') result *= currentNum; else if(lastOp == '/' && currentNum != 0) result /= currentNum;
            displayBuffer = String(result); currentNum = result; nextNumClears = true;
          } else if(val == 'C') { currentNum = 0; result = 0; lastOp = ' '; displayBuffer = "0"; nextNumClears = false; }
          delay(250);
        }
      }
    }
    display.display();
  }
  currentApp = -1; delay(300);
}

// Mantener el resto de funciones igual (runAdvancedTimer, runSettings, etc.)
void runAdvancedTimer() {
  unsigned long startT = 0, elapsed = 0; bool running = false;
  while(digitalRead(PIN_ESC) == HIGH) {
    if(running) elapsed = (millis() - startT) / 1000;
    display.clearDisplay(); display.drawRect(0, 0, 128, 64, 1);
    display.fillRect(0, 0, 128, 10, 1); display.setTextColor(0); display.setCursor(3, 1); display.print(F("STOPWATCH PRO"));
    display.setTextColor(1); display.drawCircle(30, 35, 15, 1);
    display.setTextSize(2); display.setCursor(60, 28); display.print(elapsed); display.print(F("s"));
    display.setTextSize(1); display.drawRect(60, 48, 60, 12, 1);
    display.setCursor(65, 50); display.print(running ? F("STOP") : F("START"));
    if(digitalRead(PIN_OK) == LOW) {
      if(!running) { startT = millis() - (elapsed * 1000); running = true; }
      else running = false; delay(300);
    }
    display.display();
  }
  currentApp = -1; delay(300);
}

void runSettings() {
  int subMenu = 0;
  while(digitalRead(PIN_ESC) == HIGH) {
    display.clearDisplay(); display.setCursor(0,0); display.println(F("CONFIGURACION"));
    const char* opts[] = {"ESPECIFICACIONES", "PRUEBA STICK"};
    for(int i=0; i<2; i++) {
      display.setCursor(10, 15+(i*11)); if(subMenu == i) display.print(F("> ")); display.println(opts[i]);
    }
    int vy = analogRead(JOY_Y);
    if(vy < 300) { subMenu++; if(subMenu > 1) subMenu = 0; delay(200); }
    if(vy > 700) { subMenu--; if(subMenu < 0) subMenu = 1; delay(200); }
    if(digitalRead(PIN_OK) == LOW) {
      if(subMenu == 1) { 
        while(digitalRead(PIN_ESC) == HIGH) {
          int rx = analogRead(JOY_X); int ry = analogRead(JOY_Y);
          display.clearDisplay(); display.drawCircle(map(rx,0,1023,0,128), map(ry,0,1023,64,0), 5, 1); display.display();
        }
      }
      delay(200);
    }
    display.display();
  }
  currentApp = -1; delay(300);
}
