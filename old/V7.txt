#include <Wire.h>
#include <Adafruit_GFX.h>
#include <Adafruit_SSD1306.h>
#include <Arduino.h>

#define SCREEN_WIDTH 128
#define SCREEN_HEIGHT 64
#define OLED_RESET    -1
Adafruit_SSD1306 display(SCREEN_WIDTH, SCREEN_HEIGHT, &Wire, OLED_RESET);

// Variables para el cursor (mouse) y joystick
int cursorX = SCREEN_WIDTH / 2;
int cursorY = SCREEN_HEIGHT / 2;
const int selectButtonPin = 2; // Botón para abrir apps
const int exitButtonPin = 3;   // Botón para salir de las apps
int joystickX = 0;
int joystickY = 0;
bool inApp = false; // Bandera para determinar si estamos dentro de una app
String hackCode = ""; // Para almacenar el código ingresado en la app "Hack"

void setup() {
  pinMode(selectButtonPin, INPUT_PULLUP); // Configurar botón de selección como entrada
  pinMode(exitButtonPin, INPUT_PULLUP);   // Configurar botón de salida como entrada
  pinMode(A0, INPUT); // Joystick eje X
  pinMode(A1, INPUT); // Joystick eje Y

  if (!display.begin(SSD1306_SWITCHCAPVCC, 0x3C)) {
    for (;;); // Bucle infinito si falla la pantalla
  }
  display.clearDisplay();

  // Mostrar pantalla de carga
  showLoadingScreen();

  // Mostrar íconos y el mouse
  display.clearDisplay();
  setupIconsAndText();
  drawMouse();
  display.display();
}

void showLoadingScreen() {
  String title = "ArduinoOS";
  int loadTime = random(2000, 5000); // Tiempo aleatorio entre 2 y 5 segundos
  unsigned long startTime = millis();

  display.setTextSize(2);
  display.setTextColor(SSD1306_WHITE);
  display.setCursor((SCREEN_WIDTH - 100) / 2, 5);
  display.print(title);
  display.display();

  while (millis() - startTime < loadTime) {
    display.clearDisplay();
    display.setCursor((SCREEN_WIDTH - 100) / 2, 5);
    display.print(title);
    for (int i = 0; i < 8; i++) {
      float angle = (millis() / 75.0) + (PI / 4 * i); // Velocidad ajustada
      int x = SCREEN_WIDTH / 2 + cos(angle) * 8;
      int y = 45 + sin(angle) * 8;
      display.fillCircle(x, y, 2, SSD1306_WHITE);
    }
    display.display();
    delay(30);
  }
}

void setupIconsAndText() {
  display.setTextSize(1);
  display.setCursor(19, 40); // Ícono "Calc"
  display.print("Calc");
  display.drawRect(14, 25, 30, 15, SSD1306_WHITE);

  display.setCursor(54, 40); // Ícono "Hack"
  display.print("Hack");
  display.drawRect(49, 25, 30, 15, SSD1306_WHITE);

  display.setCursor(89, 40); // Ícono "MStest"
  display.print("MStest");
  display.drawRect(84, 25, 30, 15, SSD1306_WHITE);
}

void drawMouse() {
  display.fillRect(cursorX - 2, cursorY - 2, 4, 4, SSD1306_WHITE); // Cursor
}

void checkSelection() {
  if (!inApp && digitalRead(selectButtonPin) == LOW) {
    if (cursorX >= 14 && cursorX <= 44 && cursorY >= 25 && cursorY <= 40) {
      launchApp("Calc");
    } else if (cursorX >= 49 && cursorX <= 79 && cursorY >= 25 && cursorY <= 40) {
      inApp = true; // Entramos a la app Hack
      launchHackApp();
    } else if (cursorX >= 84 && cursorX <= 114 && cursorY >= 25 && cursorY <= 40) {
      inApp = true; // Entramos a la app MStest
      launchJoystickTest();
    }
  }
}

void launchApp(const String &appName) {
  display.clearDisplay();
  display.setTextSize(2);
  display.setCursor(10, SCREEN_HEIGHT / 2 - 10);
  display.print(appName);
  display.display();
  delay(2000); // Simular la carga de la app
  setupIconsAndText(); // Volver al menú
}

void launchHackApp() {
  hackCode = ""; // Reiniciar el código ingresado
  String previousHackCode = ""; // Código anterior para comparar
  int previousCursorX = -1, previousCursorY = -1; // Posición previa del cursor para comparar

  while (true) {
    // Leer joystick para mover el cursor
    joystickX = analogRead(A0);
    joystickY = analogRead(A1);
    int deltaX = map(joystickX, 512, 1023, 0, 2) - map(joystickX, 0, 511, 2, 0);
    int deltaY = map(joystickY, 512, 1023, 0, 2) - map(joystickY, 0, 511, 2, 0);
    int newCursorX = constrain(cursorX + deltaX, 0, SCREEN_WIDTH - 1);
    int newCursorY = constrain(cursorY + deltaY, 0, SCREEN_HEIGHT - 1);

    // Detectar si el cursor se movió
    bool cursorMoved = (newCursorX != previousCursorX || newCursorY != previousCursorY);
    cursorX = newCursorX;
    cursorY = newCursorY;

    // Detectar selección de número y evitar múltiples selecciones por pulsación
    static bool buttonPressed = false; // Variable para gestionar pulsaciones únicas
    if (digitalRead(selectButtonPin) == LOW && !buttonPressed) {
      buttonPressed = true; // Marcar botón como presionado

      if (cursorX >= 15 && cursorX <= 35 && cursorY >= SCREEN_HEIGHT - 20 && cursorY <= SCREEN_HEIGHT - 5) {
        hackCode += "1";
      } else if (cursorX >= 55 && cursorX <= 75 && cursorY >= SCREEN_HEIGHT - 20 && cursorY <= SCREEN_HEIGHT - 5) {
        hackCode += "2";
      } else if (cursorX >= 95 && cursorX <= 115 && cursorY >= SCREEN_HEIGHT - 20 && cursorY <= SCREEN_HEIGHT - 5) {
        hackCode += "3";
      }
    }

    // Resetear la variable de botón cuando se suelta
    if (digitalRead(selectButtonPin) == HIGH) {
      buttonPressed = false;
    }

    // Limitar el código a 3 números
    if (hackCode.length() == 3) {
      display.clearDisplay();
      display.setTextSize(2);
      display.setCursor((SCREEN_WIDTH - (hackCode.length() * 12)) / 2, SCREEN_HEIGHT / 2 - 10); // Centrar el código
      display.print(hackCode);
      display.display();
      delay(1000); // Breve pausa antes de procesar

      if (hackCode == "21") {
        // Ejecutar barras de carga y mostrar mensaje
        displayLoadingBar(5, "Barra 1");
        displayLoadingBar(10, "Barra 2");
        displayLoadingBar(15, "Barra 3");
        displayMessage("TI DESTRUCTOR SERVER");
      } else if (hackCode == "213") {
        // Ejecutar una barra de carga y mostrar mensaje
        displayLoadingBar(8, "Cargando...");
        displayMessage("HOLA");
      }

      // Reiniciar para nuevo ingreso
      hackCode = "";
      previousHackCode = "";
    }

    // Solo redibujar si hay cambios
    if (cursorMoved || hackCode != previousHackCode) {
      display.clearDisplay();

      // Dibujar texto "Ingresa un código"
      display.setTextSize(1);
      display.setCursor(20, 10);
      display.print("Ingresa un codigo:");

      // Dibujar números en la parte inferior
      display.setCursor(20, SCREEN_HEIGHT - 15);
      display.print("1");
      display.drawRect(15, SCREEN_HEIGHT - 20, 20, 15, SSD1306_WHITE);

      display.setCursor(60, SCREEN_HEIGHT - 15);
      display.print("2");
      display.drawRect(55, SCREEN_HEIGHT - 20, 20, 15, SSD1306_WHITE);

      display.setCursor(100, SCREEN_HEIGHT - 15);
      display.print("3");
      display.drawRect(95, SCREEN_HEIGHT - 20, 20, 15, SSD1306_WHITE);

      // Dibujar el código ingresado
      display.setTextSize(2);
      display.setCursor((SCREEN_WIDTH - (hackCode.length() * 12)) / 2, SCREEN_HEIGHT / 2 - 10); // Centrar el texto
      display.print(hackCode);

      // Dibujar el cursor
      drawMouse();
      display.display();

      // Actualizar variables previas
      previousCursorX = cursorX;
      previousCursorY = cursorY;
      previousHackCode = hackCode;
    }

    // Salir al menú principal
    if (digitalRead(exitButtonPin) == LOW) {
      inApp = false;
      display.clearDisplay();
      setupIconsAndText();
      display.display();
      return;
    }

    delay(10); // Movimiento más fluido
  }
}

void displayLoadingBar(int duration, const char *message) {
  unsigned long startTime = millis();
  while (millis() - startTime < (duration * 1000)) {
    display.clearDisplay();
    display.setTextSize(1);
    display.setCursor(10, 10);
    display.print(message);

    int progress = map(millis() - startTime, 0, duration * 1000, 0, SCREEN_WIDTH);
    display.fillRect(0, SCREEN_HEIGHT / 2, progress, 10, SSD1306_WHITE);
    display.display();
    delay(50);
  }
}

void displayMessage(const char *message) {
  display.clearDisplay();
  display.setTextSize(2);
  display.setCursor((SCREEN_WIDTH - (strlen(message) * 12)) / 2, SCREEN_HEIGHT / 2 - 10); // Centrar mensaje
  display.print(message);
  display.display();
  delay(3000); // Mostrar el mensaje por 3 segundos
}

void launchJoystickTest() {
  while (true) {
    joystickX = analogRead(A0);
    joystickY = analogRead(A1);

    // Movimiento más fluido al escalar el mapeo de joystick
    int deltaX = map(joystickX, 512, 1023, 0, 4) - map(joystickX, 0, 511, 4, 0);
    int deltaY = map(joystickY, 512, 1023, 0, 4) - map(joystickY, 0, 511, 4, 0);

    cursorX = constrain(cursorX + deltaX, 0, SCREEN_WIDTH - 1);
    cursorY = constrain(cursorY + deltaY, 0, SCREEN_HEIGHT - 1);

    display.clearDisplay();
    display.fillCircle(cursorX, cursorY, 2, SSD1306_WHITE); // Representar la posición del cursor

    display.setTextSize(1);
    display.setCursor(0, 0);
    display.print("X: ");
    display.print(cursorX);
    display.print(" Y: ");
    display.print(cursorY);

    display.display();
    delay(30); // Ajuste menor para suavidad en el movimiento

    // Salir al menú principal
    if (digitalRead(exitButtonPin) == LOW) {
      inApp = false;
      display.clearDisplay();
      setupIconsAndText();
      display.display();
      return;
    }
  }
}

void loop() {
  if (!inApp) {
    joystickX = analogRead(A0);
    joystickY = analogRead(A1);

    int deltaX = map(joystickX, 512, 1023, 0, 2) - map(joystickX, 0, 511, 2, 0);
    int deltaY = map(joystickY, 512, 1023, 0, 2) - map(joystickY, 0, 511, 2, 0);

    cursorX = constrain(cursorX + deltaX, 0, SCREEN_WIDTH - 1);
    cursorY = constrain(cursorY + deltaY, 0, SCREEN_HEIGHT - 1);

    display.clearDisplay();
    setupIconsAndText();
    drawMouse();
    display.display();

    checkSelection();
    delay(10); // Movimiento más fluido en el menú
  }
}
