#include <Wire.h>
#include <Adafruit_GFX.h>
#include <Adafruit_SSD1306.h>

#define SCREEN_WIDTH 128
#define SCREEN_HEIGHT 64
Adafruit_SSD1306 display(SCREEN_WIDTH, SCREEN_HEIGHT, &Wire, -1);

// --- Configuración de Pines ---
const int PIN_OK = 2;   
const int PIN_ESC = 3;  
const int JOY_X = A0;
const int JOY_Y = A1;

// --- Variables de Sistema ---
float cursorX = 64, cursorY = 32;
int currentApp = -1; 
int selectedIcon = -1;

void setup() {
  pinMode(PIN_OK, INPUT_PULLUP);
  pinMode(PIN_ESC, INPUT_PULLUP);
  if(!display.begin(SSD1306_SWITCHCAPVCC, 0x3C)) for(;;);
  bootWin10(); 
}

void loop() {
  if (currentApp == -1) {
    updateDesktop();
    drawDesktop();
  } else {
    if (currentApp == 0) runFullCalc(); 
    if (currentApp == 1) runTimer();
    if (currentApp == 2) runStickGraph();
  }
}

// ==========================================
// 1. CARGA WINDOWS 10 & ESCRITORIO
// ==========================================
void bootWin10() {
  for(int frame = 0; frame < 12; frame++) {
    display.clearDisplay();
    display.setCursor(35, 20); display.print(F("Arduino OS"));
    for(int i=0; i<8; i++) {
      float angle = (frame * 0.4) + (i * 0.8);
      display.fillCircle(64 + cos(angle)*10, 45 + sin(angle)*10, 1, 1);
    }
    display.display(); delay(80);
  }
}

void updateDesktop() {
  int vx = analogRead(JOY_X); int vy = analogRead(JOY_Y);
  if(vx < 400) cursorX -= 3.5; else if(vx > 600) cursorX += 3.5;
  if(vy < 400) cursorY -= 3.5; else if(vy > 600) cursorY += 3.5;
  cursorX = constrain(cursorX, 0, 120); cursorY = constrain(cursorY, 0, 54);

  selectedIcon = -1;
  if (cursorY > 10 && cursorY < 35) {
    if (cursorX < 42) selectedIcon = 0;
    else if (cursorX < 84) selectedIcon = 1;
    else if (cursorX >= 84) selectedIcon = 2;
  }
  if (digitalRead(PIN_OK) == LOW && selectedIcon != -1) { currentApp = selectedIcon; delay(300); }
}

void drawDesktop() {
  display.clearDisplay();
  // Iconos Reales
  if(selectedIcon == 0) display.drawRect(8, 8, 30, 22, 1);
  display.setCursor(12, 15); display.print(F("CALC"));
  
  if(selectedIcon == 1) display.drawRect(50, 8, 30, 22, 1);
  display.setCursor(54, 15); display.print(F("TIME"));
  
  if(selectedIcon == 2) display.drawRect(92, 8, 30, 22, 1);
  display.setCursor(96, 15); display.print(F("STCK"));
  
  // Barra de Tareas
  display.fillRect(0, 56, 128, 8, 1);
  display.setTextColor(0); display.setCursor(2, 57); display.print(F("START"));
  display.setCursor(75, 57); display.print(F("Arduino OS"));
  display.setTextColor(1);
  
  // Puntero de Flecha
  display.fillTriangle(cursorX, cursorY, cursorX, cursorY+7, cursorX+5, cursorY+5, 1);
  display.display();
}

// ==========================================
// 2. APP: CALCULADORA PRO (TECLADO 0-9)
// ==========================================
void runFullCalc() {
  long result = 0, currentNum = 0;
  char lastOp = ' ';
  String displayBuffer = "0";

  struct Btn { int x, y, w, h; char val; };
  Btn b[] = {
    {5,18,18,10,'7'}, {28,18,18,10,'8'}, {51,18,18,10,'9'}, {74,18,18,10,'+'}, {97,18,18,22,'C'},
    {5,30,18,10,'4'}, {28,30,18,10,'5'}, {51,30,18,10,'6'}, {74,30,18,10,'-'},
    {5,42,18,10,'1'}, {28,42,18,10,'2'}, {51,42,18,10,'3'}, {74,42,18,10,'='}, {97,42,18,10,'0'}
  };

  while(digitalRead(PIN_ESC) == HIGH) {
    display.clearDisplay();
    display.drawRect(0, 0, 128, 14, 1); // Pantalla LCD
    display.setCursor(5, 3); display.print(displayBuffer);

    for(int i=0; i<14; i++) {
      display.drawRect(b[i].x, b[i].y, b[i].w, b[i].h, 1);
      display.setCursor(b[i].x+6, b[i].y+1); display.print(b[i].val);
    }

    int vx = analogRead(JOY_X); int vy = analogRead(JOY_Y);
    if(vx < 400) cursorX -= 3; else if(vx > 600) cursorX += 3;
    if(vy < 400) cursorY -= 3; else if(vy > 600) cursorY += 3;
    cursorX = constrain(cursorX, 0, 120); cursorY = constrain(cursorY, 0, 60);
    display.fillCircle(cursorX, cursorY, 2, 1); // Puntero pequeño para precisión
    
    if(digitalRead(PIN_OK) == LOW) {
      for(int i=0; i<14; i++) {
        if(cursorX >= b[i].x && cursorX <= b[i].x+b[i].w && cursorY >= b[i].y && cursorY <= b[i].y+b[i].h) {
          char val = b[i].val;
          if(val >= '0' && val <= '9') {
            currentNum = (currentNum * 10) + (val - '0');
            displayBuffer = String(currentNum);
          } else if(val == '+' || val == '-') {
            result = currentNum; currentNum = 0; lastOp = val; displayBuffer = String(val);
          } else if(val == '=') {
            if(lastOp == '+') result += currentNum; else if(lastOp == '-') result -= currentNum;
            displayBuffer = String(result); currentNum = result;
          } else if(val == 'C') {
            currentNum = 0; result = 0; lastOp = ' '; displayBuffer = "0";
          }
          delay(200);
        }
      }
    }
    display.display();
  }
  currentApp = -1; delay(300);
}

// ==========================================
// 3. CRONÓMETRO CON INICIO REAL
// ==========================================
void runTimer() {
  unsigned long startT = 0, elapsed = 0;
  bool running = false;
  while(digital_read_esc()) {
    if(running) elapsed = (millis() - startT) / 1000;
    display.clearDisplay();
    display.setCursor(0,0); display.print(F("STOPWATCH"));
    display.setTextSize(2); display.setCursor(35, 25); display.print(elapsed); display.print(F("s"));
    display.setTextSize(1); display.setCursor(0, 56); 
    display.print(running ? F("OK:Pausar") : F("OK:INICIAR"));
    display.display();
    if(digitalRead(PIN_OK) == LOW) {
      if(!running) { startT = millis() - (elapsed * 1000); running = true; }
      else { running = false; }
      delay(300);
    }
  }
  currentApp = -1; delay(300);
}

// ==========================================
// 4. VECTOR STICK (RADAR GRÁFICO)
// ==========================================
void runStickGraph() {
  while(digital_read_esc()) {
    int rx = analogRead(JOY_X); int ry = analogRead(JOY_Y);
    int gX = map(rx, 0, 1023, 34, 94); int gY = map(ry, 0, 1023, 50, 14);
    display.clearDisplay();
    display.setCursor(0,0); display.print(F("VECTOR RADAR"));
    display.drawRect(34, 14, 60, 36, 1); 
    display.drawLine(64, 14, 64, 50, 1); display.drawLine(34, 32, 94, 32, 1); 
    display.fillCircle(gX, gY, 4, 1);    
    display.setCursor(0, 56); display.print(F("X:")); display.print(rx); display.print(F(" Y:")); display.print(ry);
    display.display();
  }
  currentApp = -1; delay(300);
}

bool digital_read_esc() { return digitalRead(PIN_ESC) == HIGH; }
